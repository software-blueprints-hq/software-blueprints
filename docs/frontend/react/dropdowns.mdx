```jsx
import React, { useState, useRef, useEffect } from 'react';

export default function ActionDropdown({ onEdit, onDuplicate, onDelete }) {
  const [isOpen, setIsOpen] = useState(false);
  const menuRef = useRef(null);

  useEffect(() => {
    function handleClickOutside(e) {
      if (menuRef.current && !menuRef.current.contains(e.target)) {
        setIsOpen(false);
      }
    }
    document.addEventListener('mousedown', handleClickOutside);
    return () => document.removeEventListener('mousedown', handleClickOutside);
  }, []);

  return (
    <div className="dropdown-container" ref={menuRef}>
      <button 
        type="button" 
        className="dropdown-trigger"
        aria-haspopup="true"
        aria-expanded={isOpen}
        onClick={() => setIsOpen(!isOpen)}
      >
        Options <span>▾</span>
      </button>

      {isOpen && (
        <ul className="dropdown-menu" role="menu">
          <li role="menuitem" onClick={() => { onEdit(); setIsOpen(false); }}>Edit Blueprint</li>
          <li role="menuitem" onClick={() => { onDuplicate(); setIsOpen(false); }}>Duplicate Schema</li>
          <li className="dropdown-divider" role="separator"></li>
          <li role="menuitem" className="danger" onClick={() => { onDelete(); setIsOpen(false); }}>Delete Resource</li>
        </ul>
      )}
    </div>
  );
}
```

    <div className="select-wrapper">
      <label htmlFor="status-select">Deployment Status</label>
      <div className="select-container">
        <select 
          id="status-select" 
          value={value} 
          onChange={(e) => onChange(e.target.value)}
        >
          <option value="healthy">🟢 Live (Healthy)</option>
          <option value="degraded">🟡 Degraded (High Latency)</option>
          <option value="outage">🔴 Outage (Down)</option>
