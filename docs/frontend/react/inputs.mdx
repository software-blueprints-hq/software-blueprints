# Inputs & Textareas

Essential form controls designed with semantic markup, explicit labels (`<label htmlFor>`), helper text with `aria-describedby`, error state handling (`aria-invalid="true"`), and focus rings.

```jsx
import React from 'react';

export default function FormInputs({ username, dbUrl, error }) {
  return (
    <div className="input-grid">
      {/* Standard Input */}
      <div className="input-group">
        <label htmlFor="username">Username</label>
        <input 
          id="username" 
          type="text" 
          placeholder="e.g. alex_developer" 
          aria-describedby="username-hint"
        />
        <span id="username-hint" className="input-hint">Visible across public organization profiles.</span>
      </div>

      {/* Error State Input */}
      <div className="input-group has-error">
        <label htmlFor="db-url">Database URL</label>
        <input 
          id="db-url" 
          type="text" 
          defaultValue={dbUrl} 
          aria-invalid={!!error}
          aria-describedby="db-error"
        />
        {error && <span id="db-error" className="input-error">{error}</span>}
      </div>
    </div>
  );
}
```

```jsx
import React, { useState } from 'react';

export default function TextareaWithCount({ maxLength = 300 }) {
  const [text, setText] = useState('A decoupled event-driven architecture...');

  return (
    <div className="textarea-wrapper">
      <div className="textarea-header">
        <label htmlFor="arch-summary">Architecture Summary</label>
        <span className="char-count">{text.length} / {maxLength}</span>
      </div>
      <textarea
        id="arch-summary"
        rows={4}
        maxLength={maxLength}
        value={text}
        onChange={(e) => setText(e.target.value)}
      />
    </div>
  );
}
```
