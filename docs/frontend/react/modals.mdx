```jsx
import React from 'react';

export default function ConfirmationModal({ isOpen, onClose, onConfirm }) {
  if (!isOpen) return null;

  return (
    <div className="modal-backdrop" onClick={onClose}>
      <div 
        className="modal-card" 
        role="dialog" 
        aria-modal="true" 
        aria-labelledby="dialog-title"
        onClick={(e) => e.stopPropagation()}
      >
        <h3 id="dialog-title" className="modal-title">Deactivate Workspace</h3>
        <p className="modal-description">
          Are you sure you want to deactivate this workspace? All active member sessions will be terminated and scheduled jobs will be paused immediately.
        </p>
        <div className="modal-actions">
          <button type="button" className="btn-cancel" onClick={onClose}>Cancel</button>
          <button type="button" className="btn-danger" onClick={onConfirm}>Deactivate</button>
        </div>
      </div>
    </div>
  );
}
```

```jsx
import React, { useState } from 'react';

export default function CreateApiKeyModal({ isOpen, onClose, onCreate }) {
  const [label, setLabel] = useState('');
  const [scope, setScope] = useState('full');

  if (!isOpen) return null;

  const handleSubmit = (e) => {
    e.preventDefault();
    onCreate({ label, scope });
    onClose();
  };

  return (
    <div className="modal-backdrop" onClick={onClose}>
      <div 
        className="modal-card" 
        role="dialog" 
        aria-modal="true" 
        aria-labelledby="modal-api-title"
        onClick={(e) => e.stopPropagation()}
      >
        <div className="modal-header">
          <h3 id="modal-api-title">Create API Key</h3>
          <button type="button" className="btn-close" onClick={onClose} aria-label="Close modal">✕</button>
        </div>
        <form onSubmit={handleSubmit}>
          <div className="form-group">
            <label htmlFor="key-label">Key Label</label>
            <input 
              id="key-label" 
              type="text" 
              required
              value={label} 
              onChange={(e) => setLabel(e.target.value)} 
              placeholder="e.g. production-worker-alpha"
            />
          </div>
          <div className="form-group">
            <label htmlFor="key-scope">Scope</label>
            <select id="key-scope" value={scope} onChange={(e) => setScope(e.target.value)}>
              <option value="full">Full Access (Read / Write / Delete)</option>
              <option value="read">Read Only</option>
              <option value="ingest">Metrics & Ingestion Only</option>
            </select>
          </div>
          <div className="modal-actions">
            <button type="button" className="btn-cancel" onClick={onClose}>Cancel</button>
            <button type="submit" className="btn-primary">Generate Key</button>
          </div>
        </form>
      </div>
    </div>
  );
}
```

```jsx
import React from 'react';

export default function SlideDrawer({ isOpen, onClose, eventData }) {
  if (!isOpen) return null;

  return (
    <div className="drawer-backdrop" onClick={onClose}>
      <aside 
        className="drawer-panel" 
        role="dialog" 
        aria-modal="true" 
        aria-label="Event Details"
        onClick={(e) => e.stopPropagation()}
      >
        <div className="drawer-header">
          <h4>Event Details</h4>
          <button type="button" onClick={onClose} aria-label="Close drawer">✕</button>
        </div>
        <div className="drawer-content">

          <p><strong>Actor:</strong> {eventData?.actor}</p>
          <p><strong>Timestamp:</strong> {eventData?.timestamp}</p>
        </div>
        <div className="drawer-footer">
          <button type="button" className="btn-secondary">Export Raw JSON</button>
        </div>
      </aside>
    </div>
  );
}
```
